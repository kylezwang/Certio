## 61. Audit Log Summary Statistics
docker exec -it certio-sqlserver /opt/mssql-tools18/bin/sqlcmd -S localhost,1433 -U sa -P $env:SQL_PASSWORD -C -N -W -s',' -Q "SELECT 'Total Actions' AS Metric, COUNT(*) AS Count FROM CertioLocal.dbo.AuditLogs UNION ALL SELECT 'Unique Users', COUNT(DISTINCT UserId) FROM CertioLocal.dbo.AuditLogs WHERE UserId IS NOT NULL UNION ALL SELECT 'Failed Actions', COUNT(*) FROM CertioLocal.dbo.AuditLogs WHERE Result = 'FAILURE' UNION ALL SELECT 'AI Actions', COUNT(*) FROM CertioLocal.dbo.AuditLogs WHERE IsAIAction = 1 UNION ALL SELECT 'Login Attempts', COUNT(*) FROM CertioLocal.dbo.AuditLogs WHERE Action IN ('Login', 'LoginFailed') ORDER BY Metric"

## 62. Check Audit Logs by Result Status
docker exec -it certio-sqlserver /opt/mssql-tools18/bin/sqlcmd -S localhost,1433 -U sa -P $env:SQL_PASSWORD -C -N -W -s',' -Q "SELECT Result, COUNT(*) AS Count FROM CertioLocal.dbo.AuditLogs GROUP BY Result ORDER BY Count DESC"

## 63. Check Most Active Entities (Most Audited)
docker exec -it certio-sqlserver /opt/mssql-tools18/bin/sqlcmd -S localhost,1433 -U sa -P $env:SQL_PASSWORD -C -N -W -s',' -Q "SELECT EntityType, EntityId, COUNT(*) AS AuditCount FROM CertioLocal.dbo.AuditLogs GROUP BY EntityType, EntityId ORDER BY AuditCount DESC"

## 64. Check Audit Logs by Hour (Activity Pattern)
docker exec -it certio-sqlserver /opt/mssql-tools18/bin/sqlcmd -S localhost,1433 -U sa -P $env:SQL_PASSWORD -C -N -W -s',' -Q "SELECT DATEPART(hour, Timestamp) AS Hour, COUNT(*) AS ActionCount FROM CertioLocal.dbo.AuditLogs WHERE Timestamp >= DATEADD(day, -7, GETDATE()) GROUP BY DATEPART(hour, Timestamp) ORDER BY Hour"

## 65. Check Audit Logs by Day of Week (Activity Pattern)
docker exec -it certio-sqlserver /opt/mssql-tools18/bin/sqlcmd -S localhost,1433 -U sa -P $env:SQL_PASSWORD -C -N -W -s',' -Q "SELECT DATENAME(weekday, Timestamp) AS DayOfWeek, COUNT(*) AS ActionCount FROM CertioLocal.dbo.AuditLogs WHERE Timestamp >= DATEADD(day, -30, GETDATE()) GROUP BY DATENAME(weekday, Timestamp), DATEPART(weekday, Timestamp) ORDER BY DATEPART(weekday, Timestamp)"
